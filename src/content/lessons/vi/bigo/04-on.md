    def find_max(values):
        largest = values[0]
        for value in values:
            if value > largest:
                largest = value
        return largest

    print(find_max([4, 9, 2, 7]))

        int findMax(const vector<int>& values) {
            int largest = values[0];
            for (int value : values) {
                if (value > largest) largest = value;
            }
            return largest;
        }

        int main() {
            cout << findMax({4, 9, 2, 7}) << endl;
        }
