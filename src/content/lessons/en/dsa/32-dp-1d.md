    # Climbing Stairs: you can climb 1 or 2 steps at a time. How many ways to reach step n?
    # ways(n) = ways(n - 1) + ways(n - 2)  -- the same recurrence as Fibonacci.

    def ways_memo(n, memo):              # top-down: recursion + cache
        if n <= 1:
            return 1
        if n not in memo:
            memo[n] = ways_memo(n - 1, memo) + ways_memo(n - 2, memo)
        return memo[n]

    def ways_table(n):                   # bottom-up: fill a table from the base cases
        dp = [1] * (n + 1)
        for i in range(2, n + 1):
            dp[i] = dp[i - 1] + dp[i - 2]
        return dp

    def ways_two_vars(n):                # only the last two values are needed: O(1) space
        a, b = 1, 1
        for _ in range(n - 1):
            a, b = b, a + b
        return b

    print(ways_memo(10, {}), ways_table(10), ways_two_vars(10))

        function waysMemo(n, memo = new Map()) {     // top-down: recursion + cache

          if (!memo.has(n)) memo.set(n, waysMemo(n - 1, memo) + waysMemo(n - 2, memo));

        #include <iostream>
        #include <unordered_map>
        #include <vector>
        using namespace std;

        // Climbing Stairs: you can climb 1 or 2 steps at a time. How many ways to reach step n?
        // ways(n) = ways(n - 1) + ways(n - 2)  -- the same recurrence as Fibonacci.

        long long waysMemo(int n, unordered_map<int, long long>& memo) {   // top-down
            if (n <= 1) return 1;
            if (!memo.count(n)) memo[n] = waysMemo(n - 1, memo) + waysMemo(n - 2, memo);
            return memo[n];
        }

        vector<long long> waysTable(int n) {                                // bottom-up
            vector<long long> dp(n + 1, 1);
            for (int i = 2; i <= n; i++) dp[i] = dp[i - 1] + dp[i - 2];
            return dp;
        }

        long long waysTwoVars(int n) {                                      // O(1) space
            long long a = 1, b = 1;
            for (int i = 0; i < n - 1; i++) { long long c = a + b; a = b; b = c; }
            return b;
        }

        int main() {
            unordered_map<int, long long> memo;
            cout << waysMemo(10, memo) << " | ";
            for (long long v : waysTable(10)) cout << v << " ";
            cout << "| " << waysTwoVars(10) << endl;
        }

---

Climbing Stairs: you may take 1 or 2 steps; how many ways reach step n? The last move was either 1 step (from n − 1) or 2 steps (from n − 2), so `ways(n) = ways(n − 1) + ways(n − 2)`. Written as plain recursion this repeats work exponentially, exactly like naive Fibonacci.

DP removes the repetition in two equivalent ways. **Top-down (memoization):** keep the recursion but cache each answer. **Bottom-up (tabulation):** fill an array from the base cases upward, `dp[i] = dp[i − 1] + dp[i − 2]`. Both are O(n). Since each step only needs the previous two values, you can even keep just two variables: O(1) space. The recipe for any DP: define the state, write the recurrence, set the base cases, choose the fill order.
